cmake_minimum_required(VERSION 3.16)
project(mockamap)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters kdtree search)
find_package(Eigen3 REQUIRED)
include_directories(${PCL_INCLUDE_DIRS} include)

add_executable(mockamap_node src/mockamap.cpp src/maps.cpp src/perlinnoise.cpp)
ament_target_dependencies(mockamap_node rclcpp sensor_msgs pcl_conversions)
target_link_libraries(mockamap_node ${PCL_LIBRARIES} Eigen3::Eigen)
install(TARGETS mockamap_node DESTINATION lib/${PROJECT_NAME})
foreach(dir launch config params meshes)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
    install(DIRECTORY ${dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()

ament_package()
