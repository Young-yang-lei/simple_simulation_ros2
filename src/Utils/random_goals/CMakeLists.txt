cmake_minimum_required(VERSION 3.16)
project(random_goals)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(uav_utils REQUIRED)
find_package(Eigen3 REQUIRED)

add_executable(random_goals_node src/random_goals_node.cpp)
ament_target_dependencies(random_goals_node rclcpp std_msgs geometry_msgs nav_msgs visualization_msgs quadrotor_msgs uav_utils)
target_link_libraries(random_goals_node Eigen3::Eigen)
install(TARGETS random_goals_node DESTINATION lib/${PROJECT_NAME})
foreach(dir launch config params meshes)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
    install(DIRECTORY ${dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()

ament_package()
