cmake_minimum_required(VERSION 3.16)
project(rviz_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(rviz_default_plugins REQUIRED)
find_package(pluginlib REQUIRED)

find_package(Qt5 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
add_library(${PROJECT_NAME} SHARED src/pose_tool.cpp src/goal_tool.cpp src/goal_tool.h)
target_include_directories(${PROJECT_NAME} PRIVATE include)
ament_target_dependencies(${PROJECT_NAME} rclcpp geometry_msgs quadrotor_msgs visualization_msgs rviz_common rviz_rendering rviz_default_plugins pluginlib)
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugin_description.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)

install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(plugin_load_test test/plugin_load.cpp)
  ament_target_dependencies(plugin_load_test rviz_common pluginlib)
  target_link_libraries(plugin_load_test Qt5::Widgets)
  add_test(NAME rviz_plugin_load COMMAND plugin_load_test)
  set_tests_properties(rviz_plugin_load PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
endif()
ament_package()
