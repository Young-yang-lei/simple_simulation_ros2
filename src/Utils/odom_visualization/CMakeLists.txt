cmake_minimum_required(VERSION 3.16)
project(odom_visualization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)
find_package(pose_utils REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Armadillo REQUIRED)
include_directories(${ARMADILLO_INCLUDE_DIRS})

add_executable(odom_visualization src/odom_visualization.cpp)
ament_target_dependencies(odom_visualization rclcpp std_msgs geometry_msgs nav_msgs sensor_msgs visualization_msgs quadrotor_msgs pose_utils tf2 tf2_ros tf2_geometry_msgs)
target_link_libraries(odom_visualization ${ARMADILLO_LIBRARIES} pose_utils::pose_utils)
install(TARGETS odom_visualization DESTINATION lib/${PROJECT_NAME})
foreach(dir launch config params meshes)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${dir}")
    install(DIRECTORY ${dir} DESTINATION share/${PROJECT_NAME})
  endif()
endforeach()

ament_package()
