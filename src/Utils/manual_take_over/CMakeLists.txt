cmake_minimum_required(VERSION 3.16)
project(manual_take_over)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(quadrotor_msgs REQUIRED)

find_package(Eigen3 REQUIRED)
foreach(target manual_take_over manual_take_over_station)
  if(target STREQUAL "manual_take_over")
    set(source src/manual_take_over.cpp)
  else()
    set(source src/ground_station.cpp)
  endif()
  add_executable(${target} ${source})
  ament_target_dependencies(${target} rclcpp std_msgs sensor_msgs quadrotor_msgs)
  target_link_libraries(${target} Eigen3::Eigen)
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})

ament_package()
