cmake_minimum_required(VERSION 3.16)
project(pose_utils)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wno-unused-parameter)
find_package(ament_cmake REQUIRED)

find_package(Armadillo REQUIRED)
add_library(pose_utils SHARED src/pose_utils.cpp)
target_include_directories(pose_utils PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${ARMADILLO_INCLUDE_DIRS})
target_link_libraries(pose_utils ${ARMADILLO_LIBRARIES})
install(TARGETS pose_utils EXPORT pose_utilsTargets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(pose_utilsTargets HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_libraries(pose_utils)

ament_package()
